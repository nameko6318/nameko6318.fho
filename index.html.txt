    <script>
        // --- 高画質画像保存用の IndexedDB 設定 ---
        const DB_NAME = 'HighResPhotoMemoDB';
        const STORE_NAME = 'images';
        let db;

        function initDB() {
            return new Promise((resolve, reject) => {
                const request = indexedDB.open(DB_NAME, 1);
                request.onupgradeneeded = (e) => {
                    e.target.result.createObjectStore(STORE_NAME);
                };
                request.onsuccess = (e) => {
                    db = e.target.result;
                    resolve();
                };
                request.onerror = (e) => reject(e);
            });
        }

        function saveImage(blob) {
            return new Promise((resolve, reject) => {
                const transaction = db.transaction([STORE_NAME], 'readwrite');
                const store = transaction.objectStore(STORE_NAME);
                const request = store.put(blob, 'currentBg');
                request.onsuccess = () => resolve();
                request.onerror = (e) => reject(e);
            });
        }

        function loadImage() {
            return new Promise((resolve, reject) => {
                const transaction = db.transaction([STORE_NAME], 'readonly');
                const store = transaction.objectStore(STORE_NAME);
                const request = store.get('currentBg');
                request.onsuccess = (e) => resolve(e.target.result);
                request.onerror = (e) => reject(e);
            });
        }

        // --- メイン処理 ---
        document.addEventListener('DOMContentLoaded', async () => {
            const bgContainer = document.getElementById('bg-container');
            const memoText = document.getElementById('memo-text');
            const imageUpload = document.getElementById('image-upload');
            const STORAGE_KEY_TEXT = 'photoMemo_text';

            // 1. テキストの読み込みと自動保存 (LocalStorage)
            const savedText = localStorage.getItem(STORAGE_KEY_TEXT);
            if (savedText) memoText.value = savedText;

            memoText.addEventListener('input', () => {
                localStorage.setItem(STORAGE_KEY_TEXT, memoText.value);
            });

            // 2. 高画質画像の読み込みと表示 (IndexedDB)
            try {
                await initDB();
                const blob = await loadImage();
                if (blob) {
                    const objectURL = URL.createObjectURL(blob);
                    bgContainer.style.backgroundImage = `url(${objectURL})`;
                }
            } catch (err) {
                console.error("データの読み込みに失敗しました:", err);
            }

            // 3. 画像がアップロードされた時の処理（無圧縮で保存）
            imageUpload.addEventListener('change', async (event) => {
                const file = event.target.files[0];
                if (!file) return;

                try {
                    // 高画質な写真データ（Blob）をそのまま保存
                    await saveImage(file);
                    
                    // 画面に即座に反映
                    const objectURL = URL.createObjectURL(file);
                    bgContainer.style.backgroundImage = `url(${objectURL})`;
                } catch (err) {
                    alert('画像の保存に失敗しました。ファイルが破損している可能性があります。');
                    console.error(err);
                }
            });
        });
    </script>
